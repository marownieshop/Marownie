const PICKUP_ADDRESS = "4640 Hedgcoxe Rd, Plano, TX 75024"; function doGet(e) { const params = e && e.parameter ? e.parameter : {}; const address = (params.address || "").trim(); const callback = params.callback || params.jsonp || ""; let result; try { if (!address) { throw new Error("Please enter your full delivery address."); } const directions = Maps.newDirectionFinder() .setOrigin(PICKUP_ADDRESS) .setDestination(address) .setMode(Maps.DirectionFinder.Mode.DRIVING) .getDirections(); if ( !directions.routes || directions.routes.length === 0 || !directions.routes[0].legs || directions.routes[0].legs.length === 0 ) { throw new Error("We could not find a driving route for this address."); } const leg = directions.routes[0].legs[0]; const distanceMiles = leg.distance.value / 1609.344; const durationMinutes = Math.ceil(leg.duration.value / 60); let fee = null; if (distanceMiles <= 5) { fee = 5; } else if (distanceMiles <= 10) { fee = 10; } else if (distanceMiles <= 15) { fee = 15; } else if (distanceMiles <= 20) { fee = 20; } else if (distanceMiles <= 30) { fee = 30; } result = { success: true, ok: true, address: address, distanceMiles: Number(distanceMiles.toFixed(1)), distance: Number(distanceMiles.toFixed(1)), durationMinutes: durationMinutes, duration: durationMinutes, fee: fee, deliveryFee: fee, quoteRequired: fee === null, message: fee === null ? "Delivery over 30 miles requires a personalized quote." : "Estimated delivery fee: $" + fee }; } catch (error) { result = { success: false, ok: false, error: error.message || "Unable to calculate delivery." }; } const json = JSON.stringify(result); if (callback && /^[a-zA-Z_$][0-9a-zA-Z_$.[\]]*$/.test(callback)) { return ContentService .createTextOutput(callback + "(" + json + ");") .setMimeType(ContentService.MimeType.JAVASCRIPT); } return ContentService .createTextOutput(json) .setMimeType(ContentService.MimeType.JSON); }
